```tsx
// ----------------------------------------------------------------------
// Animação e Temporização Temporal no Frame Loop
// ----------------------------------------------------------------------

function GlobeAnimator({
  globe,
  durationMs,
  onProgress,
  onFinished,
}: {
  globe: any;
  durationMs?: number;
  onProgress?: (progress: number) => void;
  onFinished?: () => void;
}) {
  const hasTriggeredRef = useRef(false);

  useFrame(({ clock }) => {
    if (!globe) return;
    const material = globe.globeMaterial() as THREE.MeshPhongMaterial | null;
    if (material) {
      const t = clock.getElapsedTime();
      material.emissiveIntensity = 0.85 + Math.sin(t * 1.4) * 0.1;
    }

    if (durationMs && onProgress) {
      const elapsed = clock.getElapsedTime() * 1000;
      const progress = Math.min(elapsed / durationMs, 1.0);
      onProgress(progress);

      if (progress >= 1.0 && !hasTriggeredRef.current) {
        hasTriggeredRef.current = true;
        onFinished?.();
      }
    }
  });

  return null;
}

// ----------------------------------------------------------------------
// Hook de Inicialização e Descarte do ThreeGlobe
// ----------------------------------------------------------------------

function useGlobeLogic(globe: any, hotspots: Hotspot[], config: Required<GlobeConfig>) {
  const material = useMemo(
    () =>
      new THREE.MeshPhongMaterial({
        color: new THREE.Color(config.globeColor),
        emissive: new THREE.Color('#0f172a'),
        shininess: 40,
        transparent: true,
        opacity: 0.96,
      }),
    [config.globeColor]
  );

  useEffect(() => {
    if (!globe) return () => {};
    globe.globeMaterial(material);
    return () => {
      material.dispose();
    };
  }, [globe, material]);

  useEffect(() => {
    if (!globe) return () => {};

    const geo = countries as unknown as GeoFeatureCollection;

    globe
      .hexPolygonsData(geo.features)
      .hexPolygonResolution(3)
      .hexPolygonMargin(0.6)
      .hexPolygonColor(() => config.polygonColor);

    globe
      .htmlElementsData(hotspots)
      .htmlLat((d: object) => (d as Hotspot).lat)
      .htmlLng((d: object) => (d as Hotspot).lng)
      .htmlAltitude(0.02)
      .htmlElement((d: object) => createHexNode((d as Hotspot).color ?? '#60a5fa'));

    globe
      .ringsData(hotspots)
      .ringLat((d: object) => (d as Hotspot).lat)
      .ringLng((d: object) => (d as Hotspot).lng)
      .ringColor(() => (t: number) => `rgba(96,165,250,${1 - t})`)
      .ringMaxRadius(config.maxRings)
      .ringPropagationSpeed(VISUAL_CONFIG.ringSpeed)
      .ringRepeatPeriod((d: object) => 1200 + seededRandom((d as Hotspot).lat) * 800);

    return () => {
      globe.ringsData([]).hexPolygonsData([]).htmlElementsData([]);
    };
  }, [globe, hotspots, config]);
}
```

```tsx
import type { VaultStatus } from 'src/platform/tauri/contracts/vault-contracts';

import React, {
  useRef,
  useState,
  useEffect,
  useContext,
  useCallback,
  createContext,
} from 'react';

import Box from '@mui/material/Box';
import Card from '@mui/material/Card';
import Alert from '@mui/material/Alert';
import Button from '@mui/material/Button';
import Typography from '@mui/material/Typography';
import CircularProgress from '@mui/material/CircularProgress';

import { isTauriDesktop } from 'src/platform/tauri/core/invoke';
import { VaultBridge } from 'src/platform/tauri/vault/vault-bridge';
import { WindowBridge } from 'src/platform/tauri/window/window-bridge';

import { SplashScreen } from 'src/shared/components/loading-screen';

// ----------------------------------------------------------------------
// Tipos e Estados Estritos da Máquina de Inicialização
// ----------------------------------------------------------------------

export type BootstrapState =
  | 'INITIALIZING' // Handshake e validação SQLite em andamento
  | 'READY_UNLOCKED' // Desktop: cofre inicializado e desbloqueado (acesso total)
  | 'READY_LOCKED' // Desktop: cofre inicializado mas trancado (requer senha mestre)
  | 'UNINITIALIZED' // Desktop: primeiro acesso, requer criação de senha mestre
  | 'WEB_PREVIEW' // Navegador: ambiente web sem cofre nativo de hardware
  | 'TIMEOUT_WARNING' // UX: inicialização excedeu o limite de espera (4.5s)
  | 'FATAL_ERROR'; // Falha física de I/O ou banco SQLite indisponível

export interface BootstrapContextValue {
  state: BootstrapState;
  vaultStatus: VaultStatus | null;
  errorMessage: string | null;
  retry: () => void;
  unlock: (password: string) => Promise<boolean>;
  lock: () => Promise<void>;
  refreshStatus: () => Promise<VaultStatus | null>;
}

const BootstrapContext = createContext<BootstrapContextValue | null>(null);

export function useBootstrap(): BootstrapContextValue {
  const context = useContext(BootstrapContext);
  if (!context) {
    throw new Error('useBootstrap deve ser utilizado dentro de um <AppBootstrapProvider />');
  }
  return context;
}

// ----------------------------------------------------------------------
// Provedor de Bootstrap com Proteção Estrita Contra Race Conditions
// ----------------------------------------------------------------------

type Props = {
  children: React.ReactNode;
  timeoutMs?: number;
  enableThreeAnimation?: boolean;
};

export function AppBootstrapProvider({
  children,
  timeoutMs = 4500,
  enableThreeAnimation = false,
}: Props) {
  const [state, setState] = useState<BootstrapState>('INITIALIZING');
  const [vaultStatus, setVaultStatus] = useState<VaultStatus | null>(null);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);

  // Controle sequencial estrito de tentativas (elimina Casos A e B de respostas atrasadas)
  const attemptSeqRef = useRef(0);
  const activeAttemptIdRef = useRef<number | null>(null);
  const timerRef = useRef<ReturnType<typeof setTimeout> | null>(null);

  // Portão Duplo de Inicialização: IPC Ready + Animation Done
  const isAnimationDoneRef = useRef<boolean>(!enableThreeAnimation);
  const pendingTargetStateRef = useRef<BootstrapState | null>(null);

  const handleAnimationComplete = useCallback(() => {
    isAnimationDoneRef.current = true;
    if (pendingTargetStateRef.current) {
      const next = pendingTargetStateRef.current;
      pendingTargetStateRef.current = null;
      setState(next);
    }
  }, []);

  const runBootstrap = useCallback(() => {
    // 1. Limpa timer ativo da tentativa anterior
    if (timerRef.current) {
      clearTimeout(timerRef.current);
      timerRef.current = null;
    }

    // 2. Incrementa ID de tentativa e define como tentativa ativa
    const attemptId = ++attemptSeqRef.current;
    activeAttemptIdRef.current = attemptId;

    // Reseta portão duplo e estado de janela para nova tentativa
    WindowBridge.reset();
    isAnimationDoneRef.current = !enableThreeAnimation;
    pendingTargetStateRef.current = null;

    setState('INITIALIZING');
    setErrorMessage(null);

    const notifySplashStage = (stageText: string, targetP?: number) => {
      if (typeof window !== 'undefined') {
        const updater = (
          window as unknown as {
            __DEVOS_UPDATE_SPLASH_STAGE__?: (text: string, p?: number) => void;
          }
        ).__DEVOS_UPDATE_SPLASH_STAGE__;
        if (typeof updater === 'function') {
          updater(stageText, targetP);
        }
      }
    };

    notifySplashStage('VALIDANDO AMBIENTE DE SEGURANÇA...', 20);

    // 3. Verificação de ambiente: se não for Tauri Desktop, transiciona diretamente para WEB_PREVIEW
    if (!isTauriDesktop()) {
      activeAttemptIdRef.current = null;
      if (isAnimationDoneRef.current) {
        setState('WEB_PREVIEW');
      } else {
        pendingTargetStateRef.current = 'WEB_PREVIEW';
      }
      return;
    }

    // 4. Timer de UX (4.5s): invalida a tentativa ativa IMEDIATAMENTE no timeout e interrompe animação
    timerRef.current = setTimeout(() => {
      if (activeAttemptIdRef.current === attemptId) {
        // Regra fiduciária: a tentativa expirada é invalidada aqui (impede Caso A)
        activeAttemptIdRef.current = null;
        pendingTargetStateRef.current = null;
        isAnimationDoneRef.current = true;
        setState('TIMEOUT_WARNING');
      }
    }, timeoutMs);

    // 5. Execução do handshake IPC seguro com o motor Rust
    notifySplashStage('VERIFICANDO DISPONIBILIDADE DO COFRE LOCAL...', 45);

    VaultBridge.getStatus()
      .then((status) => {
        // Verificação estrita: se a tentativa ativa não for EXATAMENTE este attemptId, descarta
        if (activeAttemptIdRef.current !== attemptId) {
          // Descarte seguro de resposta atrasada (impede Casos A e B)
          return;
        }

        // Tentativa válida no prazo
        activeAttemptIdRef.current = null;
        if (timerRef.current) {
          clearTimeout(timerRef.current);
          timerRef.current = null;
        }

        setVaultStatus(status);
        notifySplashStage('VERIFICANDO CONECTIVIDADE COM A REDE IPFS...', 75);

        // Invariante de Segurança Fiduciária — Tratamento Exaustivo das 4 Combinações:
        let targetState: BootstrapState;
        if (!status.is_initialized && status.is_unlocked) {
          // Combinação Anômala: desbloqueado sem estar inicializado
          setErrorMessage('Inconsistência de integridade na sessão do cofre.');
          targetState = 'FATAL_ERROR';
        } else if (!status.is_initialized) {
          targetState = 'UNINITIALIZED';
        } else if (!status.is_unlocked) {
          targetState = 'READY_LOCKED';
        } else {
          targetState = 'READY_UNLOCKED';
        }

        // Falha crítica interrompe imediatamente a animação
        if (targetState === 'FATAL_ERROR') {
          pendingTargetStateRef.current = null;
          isAnimationDoneRef.current = true;
          setState('FATAL_ERROR');
          return;
        }

        notifySplashStage('PREPARANDO AMBIENTE DE OPERAÇÃO...', 90);

        // Portão Duplo: Se a animação já concluiu, transiciona para o estado alvo
        if (isAnimationDoneRef.current) {
          setState(targetState);
        } else {
          // Aguarda conclusão da animação (tempo regular ou Esc/Pular)
          pendingTargetStateRef.current = targetState;
        }
      })
      .catch((err: unknown) => {
        if (activeAttemptIdRef.current !== attemptId) {
          // Descarte de erro atrasado de tentativa obsoleta
          return;
        }

        activeAttemptIdRef.current = null;
        pendingTargetStateRef.current = null;
        isAnimationDoneRef.current = true;
        if (timerRef.current) {
          clearTimeout(timerRef.current);
          timerRef.current = null;
        }

        const raw = (err as Error)?.message || String(err);
        const sanitized = raw.includes('VAULT_STORAGE_ERR')
          ? 'Falha de acesso ao armazenamento seguro local (devos_vault.sqlite).'
          : 'Erro inesperado na inicialização do subsistema nativo do cofre.';

        setErrorMessage(sanitized);
        setState('FATAL_ERROR');
      });
  }, [timeoutMs, enableThreeAnimation]);

  useEffect(() => {
    runBootstrap();

    return () => {
      // Cleanup no desmonte: anula tentativa ativa e limpa timers
      activeAttemptIdRef.current = null;
      if (timerRef.current) {
        clearTimeout(timerRef.current);
        timerRef.current = null;
      }
    };
  }, [runBootstrap]);

  // Restauração da Janela: as telas de inicialização e credenciais (UNINITIALIZED / READY_LOCKED)
  // permanecem em tela cheia fluida total (100vw x 100vh), exatamente como a tela de carregamento (splash).
  // Apenas após a liberação do cofre (READY_UNLOCKED) é que a janela é restaurada para as dimensões de trabalho.
  useEffect(() => {
    if (state === 'UNINITIALIZED' || state === 'READY_LOCKED') {
      void WindowBridge.setFullscreen(true);
    } else if (state === 'READY_UNLOCKED') {
      void WindowBridge.restoreWindowAfterBoot();
    }
  }, [state]);

  // Transição suave e descarte da 1ª página de carregamento nativa (index.html)
  useEffect(() => {
    if (state !== 'INITIALIZING') {
      const finishSplash = (window as unknown as { __DEVOS_FINISH_SPLASH__?: () => void })
        .__DEVOS_FINISH_SPLASH__;
      if (typeof finishSplash === 'function') {
        finishSplash();
      }
    }
  }, [state]);

  const unlock = useCallback(async (password: string): Promise<boolean> => {
    const ok = await VaultBridge.unlock(password);
    if (ok) {
      setState('READY_UNLOCKED');
      return true;
    }
    return false;
  }, []);

  const lock = useCallback(async (): Promise<void> => {
    try {
      await VaultBridge.lock();
    } finally {
      setState('READY_LOCKED');
    }
  }, []);

  const refreshStatus = useCallback(async (): Promise<VaultStatus | null> => {
    try {
      const status = await VaultBridge.getStatus();
      setVaultStatus(status);
      if (!status.is_initialized && status.is_unlocked) {
        setErrorMessage('Inconsistência de integridade na sessão do cofre.');
        setState('FATAL_ERROR');
      } else if (!status.is_initialized) {
        setState('UNINITIALIZED');
      } else if (!status.is_unlocked) {
        setState('READY_LOCKED');
      } else {
        setState('READY_UNLOCKED');
      }
      return status;
    } catch {
      setState('READY_LOCKED');
      return null;
```

```ts
import { invoke, isTauri } from '@tauri-apps/api/core';

/**
 * ASPPIBRA DevOS — Native IPC Invoker (Tauri v2)
 *
 * Encapsula chamadas ao motor nativo Rust com verificação de ambiente,
 * tipagem estrita de retorno e tratamento padronizado de exceções.
 */

export function isTauriDesktop(): boolean {
  return isTauri();
}

export async function invokeNative<T>(cmd: string, args: Record<string, unknown> = {}): Promise<T> {
  if (isTauriDesktop()) {
    try {
      return await invoke<T>(cmd, args);
    } catch (err: unknown) {
      const msg = typeof err === 'string' ? err : (err as Error)?.message || JSON.stringify(err);
      throw new Error(`[Rust Core IPC Error] (${cmd}): ${msg}`);
    }
  }

  throw new Error(
    `Ambiente não suportado: O comando nativo '${cmd}' exige execução dentro do aplicativo Desktop DevOS (Tauri v2).`
  );
}
```

```ts
  static async isFullscreen(): Promise<boolean> {

    try {
      const appWindow = getCurrentWindow();
      return await appWindow.isFullscreen();
    } catch {
      return false;
    }
  }

   * Define o estado de tela cheia da janela nativa.
   */
  static async setFullscreen(fullscreen: boolean): Promise<boolean> {

    try {
      if (fullscreen) {
        this.hasRestored = false;
      }
      const appWindow = getCurrentWindow();
      const current = await appWindow.isFullscreen().catch(() => null);
      if (current === fullscreen) {
        return fullscreen;
      }
      await appWindow.setFullscreen(fullscreen);
      if (!fullscreen) {
        setTimeout(async () => {
          try {
            await appWindow.setSize(
              new LogicalSize(this.DEFAULT_WINDOW_SIZE.width, this.DEFAULT_WINDOW_SIZE.height)
            );
            await appWindow.center();
          } catch {
            // ignore
          }
        }, 80);
      }
      return fullscreen;
    } catch (err: unknown) {
      console.warn('[WindowBridge] Falha ao definir tela cheia:', err);
      return false;
    }
  }

   * Alterna entre modo tela cheia e modo janela normal.
   */
  static async toggleFullscreen(): Promise<boolean> {
    if (!isTauriDesktop()) return false;
    try {
      const appWindow = getCurrentWindow();
      const current = await appWindow.isFullscreen();
      const next = !current;
      await appWindow.setFullscreen(next);
      if (!next) {
        setTimeout(async () => {
          try {
            await appWindow.setSize(
              new LogicalSize(this.DEFAULT_WINDOW_SIZE.width, this.DEFAULT_WINDOW_SIZE.height)
            );
            await appWindow.center();
          } catch {
            // ignore
          }
        }, 80);
      }
      return next;
    } catch {
      try {

        await appWindow.setFullscreen(false);
```

```ts
export type IpcErrorCode =
  | 'VALIDATION_ERROR'
  | 'BUSINESS_ERROR'
  | 'STORAGE_ERROR'
  | 'CRYPTO_ERROR'
  | 'NOT_FOUND_ERROR'
  | 'IPC_TRANSPORT_ERROR'
  | 'PERMISSION_DENIED_ERROR'
  | 'UNKNOWN_ERROR';

export interface SerializedIpcError {
  code: IpcErrorCode;
  message: string;
  command?: string;
  timestamp: string;
}

export class IpcError extends Error {
  public readonly code: IpcErrorCode;
  public readonly command?: string;
  public readonly timestamp: string;

  constructor(message: string, code: IpcErrorCode = 'UNKNOWN_ERROR', command?: string) {
    const sanitizedMessage = IpcError.sanitizeMessage(message);
    super(sanitizedMessage);
    this.name = 'IpcError';
    this.code = code;
    this.command = command;
    this.timestamp = new Date().toISOString();
  }

  /**
   * Sanitiza a mensagem para assegurar que segredos, senhas ou mnemônicas
   * nunca sejam vazados em telas ou logs de erro.
   */
  public static sanitizeMessage(raw: string): string {
    if (!raw) return 'Erro desconhecido na comunicação nativa.';
    // Remove possíveis sequências de palavras mnemônicas, seeds, chaves privadas ou senhas
    return raw
      .replace(/\b(?:mnemonic|phrase|seed|private_key|words?)\b\s*[:=]\s*[^\n,]+/gi, '[REDACTED_SECRET]')
      .replace(/\b(?:password|master_password|secret|key)\b\s*[:=]\s*[^\n,]+/gi, '[REDACTED_CREDENTIAL]');
  }

  public toJSON(): SerializedIpcError {
    return {
      code: this.code,
      message: this.message,
      command: this.command,
      timestamp: this.timestamp,
    };
  }
}
```

```ts
import { IpcError, type IpcErrorCode } from '../contracts/ipc-errors';

export class TauriErrorMapper {
  /**
   * Normaliza qualquer erro capturado do Tauri IPC em uma instância segura de IpcError.
   */
  public static normalize(error: unknown, command?: string): IpcError {
    if (error instanceof IpcError) {
      return error;
    }

    const rawMessage = typeof error === 'string'
      ? error
      : error instanceof Error
        ? error.message
        : JSON.stringify(error);

    const code = TauriErrorMapper.classifyErrorCode(rawMessage);
    return new IpcError(rawMessage, code, command);
  }

  private static classifyErrorCode(message: string): IpcErrorCode {
    const lower = message.toLowerCase();

    if (lower.includes('validation') || lower.includes('invalido') || lower.includes('invalid') || lower.includes('length')) {
      return 'VALIDATION_ERROR';
    }
    if (lower.includes('not found') || lower.includes('nao encontrado') || lower.includes('inexistente')) {
      return 'NOT_FOUND_ERROR';
    }
    if (lower.includes('permission') || lower.includes('permissão') || lower.includes('denied') || lower.includes('not allowed')) {
      return 'PERMISSION_DENIED_ERROR';
    }
    if (lower.includes('storage') || lower.includes('sqlite') || lower.includes('database') || lower.includes('banco')) {
      return 'STORAGE_ERROR';
    }
    if (lower.includes('crypto') || lower.includes('cipher') || lower.includes('argon') || lower.includes('decrypt')) {
      return 'CRYPTO_ERROR';
    }
    if (lower.includes('ipc') || lower.includes('channel') || lower.includes('webview') || lower.includes('network')) {
      return 'IPC_TRANSPORT_ERROR';
    }

    return 'BUSINESS_ERROR';
  }
}
```

```rust
fn is_marginal_whitespace_or_invisible(c: char) -> bool {
    c.is_whitespace() || c == '\u{FEFF}' || c == '\u{200B}' || c == '\u{00A0}'
}

#[tauri::command]
pub fn init_master_password(
    db: State<'_, LocalDatabase>,
    master_password: String,
) -> Result<bool, String> {
    // 1. Encapsulamento IMEDIATO na entrada: qualquer retorno de erro ou término de escopo zera a memória RAM do heap
    let master_password = zeroize::Zeroizing::new(master_password);

    // 2. Contagem rigorosa de caracteres Unicode (Code Points)
    let char_count = master_password.chars().count();
    if char_count < 12 {
        return Err("VAULT_ERR_INVALID_LENGTH: A senha mestre deve possuir pelo menos 12 caracteres para garantir a segurança criptográfica institucional.".to_string());
    }
    if char_count > 128 {
        return Err(
            "VAULT_ERR_INVALID_LENGTH: A senha mestre não pode exceder 128 caracteres.".to_string(),
        );
    }

    // 3. Verificação estrita de qualquer caractere Unicode Whitespace ou invisível nas extremidades (inclui U+FEFF, U+200B, U+00A0)
    let starts_with_whitespace = master_password
        .chars()
        .next()
        .map(is_marginal_whitespace_or_invisible)
        .unwrap_or(false);
    let ends_with_whitespace = master_password
        .chars()
        .last()
        .map(is_marginal_whitespace_or_invisible)
        .unwrap_or(false);
    if starts_with_whitespace || ends_with_whitespace {
        return Err("VAULT_ERR_MARGINAL_WHITESPACE: A senha mestre não pode conter espaços, quebras de linha ou caracteres invisíveis no início ou no fim.".to_string());
    }

    // 4. Obtenção do banco gerenciado e VERIFICAÇÃO ANTECIPADA (Fail-Fast):
    // Impede o custo computacional e de memória do Argon2id (64 MB) caso o cofre já esteja inicializado
    if db
        .get_header()
        .map_err(|e| format!("VAULT_ERR_STORAGE: {}", e))?
        .is_some()
    {
        return Err(
            "VAULT_ERR_ALREADY_INITIALIZED: O cofre já foi inicializado anteriormente.".to_string(),
        );
    }

    // 5. Criação do envelope canário com KDF pesado (Argon2id + AES-256-GCM) somente se o cofre estiver livre
    let header =
        VaultHeader::new(&master_password).map_err(|e| format!("VAULT_ERR_CRYPTO: {}", e))?;

    // 6. Persistência atômica definitiva com bloqueio imediato SQLite (garantia final contra condições de corrida)
    db.initialize_vault_if_absent(&header)
        .map_err(|e| match e {
            crate::storage::local_database::StorageError::AlreadyInitialized => {
                "VAULT_ERR_ALREADY_INITIALIZED: O cofre já foi inicializado anteriormente."
                    .to_string()
            }
            _ => format!("VAULT_ERR_STORAGE: {}", e),
        })?;

    Ok(true)
}

#[tauri::command]
pub fn unlock_vault(
    db: State<'_, LocalDatabase>,
    session: State<'_, VaultSession>,
    master_password: String,
) -> Result<bool, String> {
    // 1. Encapsulamento IMEDIATO na entrada: descarte e zeragem segura de memória (zeroize) em qualquer saída de escopo
    let master_password = zeroize::Zeroizing::new(master_password);

    // 2. Captura a época atômica no início da operação de validação da senha
    let epoch_at_start = session.current_epoch();

    let header = db
        .get_header()
        .map_err(|e| {
            log::error!(
                "[Vault Security] Erro ao recuperar cabeçalho para desbloqueio: {}",
                e
            );
            "VAULT_STORAGE_ERR: Falha ao ler cabeçalho do cofre.".to_string()
        })?
        .ok_or_else(|| "Cofre não inicializado. Crie a senha mestre primeiro.".to_string())?;

    if header.verify_password(&master_password) {
        // 3. Validação atômica de concorrência: se lock() foi acionado durante verify_password, aborta o desbloqueio
        if session.try_unlock(epoch_at_start) {
            Ok(true)
        } else {
            Err(
                "Operação de desbloqueio abortada: cofre foi bloqueado concorrentemente."
                    .to_string(),
            )
        }
    } else {
        Err("Senha mestre incorreta.".to_string())
    }
}
```

```rust
        return Err("Senha mestre inválida. Operação de custódia cancelada.".to_string());

    let mut mnemonic = if let Some(ref sid) = request.session_id {

        zeroize::Zeroizing::new(m.clone())
    } else {
        request.master_password.zeroize();
        return Err("Nenhuma sessão de criação ativa ou mnemônica informada.".to_string());

        mnemonic.zeroize();

        return Err("Frase mnemônica inválida.".to_string());

        VaultCrypto::encrypt(&request.master_password, mnemonic.as_bytes()).map_err(|e| {
            mnemonic.zeroize();

            format!("Falha ao cifrar o envelope: {}", e)
        })?;

    let words_count = mnemonic.split_whitespace().count();

    mnemonic.zeroize();
    request.master_password.zeroize();

    // 4. Cria ID de cofre independente e segregado (DAO vs Associado)

        target: request.target.clone(),

        created_at: now_iso.clone(),
```

```rust
use std::sync::Mutex;

#[derive(Default, Debug)]
struct SessionState {
    unlocked: bool,
    epoch: u64,
    session_key: Option<zeroize::Zeroizing<String>>,
}

/// Gerenciador autoritativo da sessão do cofre em memória com garantia de exclusão mútua total
#[derive(Default)]
pub struct VaultSession {
    state: Mutex<SessionState>,
}

impl VaultSession {
    pub fn new() -> Self {
        Self {
            state: Mutex::new(SessionState {
                unlocked: false,
                epoch: 0,
                session_key: None,
            }),
        }
    }

    pub fn is_unlocked(&self) -> bool {
        match self.state.lock() {
            Ok(s) => s.unlocked,
            Err(poisoned) => {
                log::warn!(
                    "[Vault Security] Mutex envenenado detectado em is_unlocked(). Aplicando Fail-Closed: cofre considerado bloqueado."
                );
                let mut s = poisoned.into_inner();
                s.unlocked = false;
                s.session_key = None;
                false
            }
        }
    }

    /// Obtém a época atômica atual da sessão para validação de operações concorrentes
    pub fn current_epoch(&self) -> u64 {
        match self.state.lock() {
            Ok(s) => s.epoch,
            Err(poisoned) => poisoned.into_inner().epoch,
        }
    }

    /// Bloqueia a sessão imediatamente, incrementa a época e purga qualquer material sensível da RAM (Poison Recovery / Fail-Closed)
    pub fn lock(&self) {
        let mut s = match self.state.lock() {
            Ok(guard) => guard,
            Err(poisoned) => {
                log::warn!(
                    "[Vault Security] Mutex envenenado detectado em lock(). Forçando recuperação para estado bloqueado."
                );
                poisoned.into_inner()
            }
        };
        s.unlocked = false;
        s.epoch = s.epoch.wrapping_add(1);
        // Descarta e zera qualquer chave de sessão sensível da RAM imediatamente via ZeroizeOnDrop:
        s.session_key = None;
    }

    /// Desbloqueia a sessão SOMENTE se a época não foi alterada por um bloqueio concorrente.
    /// Em caso de Mutex envenenado por pânico anterior, recusa imediatamente o desbloqueio (Fail-Closed).
    pub fn try_unlock(&self, expected_epoch: u64) -> bool {
        match self.state.lock() {
            Ok(mut s) => {
                if s.epoch != expected_epoch {
                    return false;
                }
                s.unlocked = true;
                true
            }
            Err(poisoned) => {
                log::error!(
                    "[Vault Security] Mutex envenenado em try_unlock(). Abortando desbloqueio com Fail-Closed."
                );
                let mut s = poisoned.into_inner();
                s.unlocked = false;
                s.session_key = None;
                false
            }
        }
    }

    /// Instala condicionalmente um recurso sensível de sessão SOMENTE se a época não mudou durante a validação.
    /// Se a época mudou (bloqueio concorrente), a transição é REJEITADA e o parâmetro `key` é descartado
    /// e imediatamente zerado na memória RAM pelo seu destrutor RAII (ZeroizeOnDrop), NUNCA sendo instalado.
    pub fn try_unlock_with_session_key(
        &self,
        expected_epoch: u64,
        key: zeroize::Zeroizing<String>,
    ) -> bool {
        let mut s = match self.state.lock() {
            Ok(guard) => guard,
            Err(poisoned) => {
                log::error!(
                    "[Vault Security] Mutex envenenado em try_unlock_with_session_key(). Abortando com Fail-Closed."
                );
                let mut s = poisoned.into_inner();
                s.unlocked = false;
                s.session_key = None;
                // `key` sai de escopo e é destruído/zerado pelo RAII
                return false;
            }
        };

        if s.epoch != expected_epoch {
            // Rejeição fiduciária: época obsoleta detectada.
            // O recurso `key` NÃO é instalado em `s.session_key` e é destruído na saída do escopo.
            return false;
        }

        s.unlocked = true;
        s.session_key = Some(key);
        true
    }

    /// Verifica se a sessão ativa possui uma chave criptográfica instalada
    pub fn has_session_key(&self) -> bool {
        match self.state.lock() {
            Ok(s) => s.session_key.is_some(),
            Err(_) => false,
        }
    }

    /// Define o estado diretamente (compatibilidade de testes)
    pub fn set_unlocked(&self, unlocked: bool) {
        let mut s = match self.state.lock() {
            Ok(guard) => guard,
            Err(poisoned) => poisoned.into_inner(),
        };
        s.unlocked = unlocked;
        if !unlocked {
            s.session_key = None;
        }
    }
}

impl VaultHeader {

    pub fn new(password: &str) -> Result<Self, VaultCryptoError> {
        let canary_envelope = VaultCrypto::encrypt(password, Self::CANARY_PLAINTEXT)?;
        let now = chrono::Utc::now().to_rfc3339();

        Ok(Self {
            version: 1,
            canary_envelope,
            created_at: now,
        })
    }

    pub fn verify_password(&self, password: &str) -> bool {
        match VaultCrypto::decrypt(password, &self.canary_envelope) {
            Ok(decrypted) => {
                let decrypted = zeroize::Zeroizing::new(decrypted);
                *decrypted == Self::CANARY_PLAINTEXT
            }
            Err(_) => false,
        }
    }
```

```rust
    pub kdf: String,

/// Chave criptográfica temporária protegida com Zeroize ao ser descartada da memória
#[derive(Zeroize, ZeroizeOnDrop)]
struct DerivedKey([u8; 32]);

pub struct VaultCrypto;

impl VaultCrypto {
    /// Deriva uma chave simétrica de 256 bits a partir da senha e do salt usando Argon2id
    fn derive_key(password: &str, salt: &[u8]) -> Result<DerivedKey, VaultCryptoError> {
        let params = Params::new(64 * 1024, 3, 1, Some(32))
            .map_err(|e| VaultCryptoError::EncryptionFailed(format!("Argon2 params: {}", e)))?;

        let argon2 = Argon2::new(Algorithm::Argon2id, Version::V0x13, params);
        let mut key_bytes = [0u8; 32];

        argon2
            .hash_password_into(password.as_bytes(), salt, &mut key_bytes)
            .map_err(|e| VaultCryptoError::EncryptionFailed(format!("Argon2 KDF: {}", e)))?;

        Ok(DerivedKey(key_bytes))
    }

    /// Cifra dados sensíveis utilizando Argon2id + AES-256-GCM
    pub fn encrypt(
        password: &str,
        plaintext: &[u8],
    ) -> Result<EncryptedEnvelope, VaultCryptoError> {
        let mut rng = rand::thread_rng();

        // Salt de 16 bytes para KDF
        let mut salt = [0u8; 16];
        rng.fill_bytes(&mut salt);

        // Nonce de 12 bytes (96 bits padrão para GCM)
        let mut nonce_bytes = [0u8; 12];
        rng.fill_bytes(&mut nonce_bytes);

        let derived_key = Self::derive_key(password, &salt)?;
        let cipher = Aes256Gcm::new_from_slice(&derived_key.0)
            .map_err(|e| VaultCryptoError::EncryptionFailed(e.to_string()))?;

        let nonce = Nonce::from_slice(&nonce_bytes);
        let ciphertext = cipher
            .encrypt(nonce, plaintext)
            .map_err(|e| VaultCryptoError::EncryptionFailed(e.to_string()))?;

        Ok(EncryptedEnvelope {
            version: 1,
            kdf: "Argon2id".to_string(),
            cipher: "AES-256-GCM".to_string(),
            salt_hex: hex::encode(salt),
            nonce_hex: hex::encode(nonce_bytes),
            ciphertext_hex: hex::encode(ciphertext),
        })
    }

    /// Decifra e autentica o envelope criptográfico. Falha caso a senha esteja incorreta ou os dados tenham sido alterados.
    pub fn decrypt(
        password: &str,
        envelope: &EncryptedEnvelope,
    ) -> Result<Vec<u8>, VaultCryptoError> {
        if envelope.version != 1 {
            return Err(VaultCryptoError::DecryptionFailed);
        }

        let salt = hex::decode(&envelope.salt_hex)
            .map_err(|e| VaultCryptoError::HexError(e.to_string()))?;
        let nonce_bytes = hex::decode(&envelope.nonce_hex)
            .map_err(|e| VaultCryptoError::HexError(e.to_string()))?;
        let ciphertext = hex::decode(&envelope.ciphertext_hex)
            .map_err(|e| VaultCryptoError::HexError(e.to_string()))?;

        let derived_key = Self::derive_key(password, &salt)?;
        let cipher = Aes256Gcm::new_from_slice(&derived_key.0)
            .map_err(|_| VaultCryptoError::DecryptionFailed)?;

        let nonce = Nonce::from_slice(&nonce_bytes);
        cipher
            .decrypt(nonce, ciphertext.as_ref())
            .map_err(|_| VaultCryptoError::DecryptionFailed)
    }
}
```

```rust
    pub fn get_secret_by_vault_id(
        &self,
        vault_id: &str,
    ) -> Result<Option<StoredSecret>, StorageError> {
        let conn = self.get_connection()?;
        let mut stmt = conn.prepare(
            "SELECT vault_id, target, envelope, created_at FROM vault_secrets WHERE vault_id = ?1",
        )?;

        let secret = stmt
            .query_row(params![vault_id], |row| {
                let v_id: String = row.get(0)?;
                let target: String = row.get(1)?;
                let env_json: String = row.get(2)?;
                let created_at: String = row.get(3)?;
                let envelope: EncryptedEnvelope = serde_json::from_str(&env_json).map_err(|e| {
                    rusqlite::Error::FromSqlConversionFailure(
                        2,
                        rusqlite::types::Type::Text,
                        Box::new(e),
                    )
                })?;

                Ok(StoredSecret {
                    vault_id: v_id,
                    target,
                    envelope,
                    created_at,
                })
            })
            .optional()?;

        Ok(secret)
    }

    /// Retorna o status operacional do cofre SQLite
    pub fn get_status(&self, is_unlocked: bool) -> Result<VaultStatus, StorageError> {
        let is_initialized = self.get_header()?.is_some();
        let wallets = self.list_wallets()?;
        let dao_count = wallets.iter().filter(|w| w.target == "DAO").count();
        let client_count = wallets.iter().filter(|w| w.target == "CLIENT").count();

        Ok(VaultStatus {
            is_initialized,
            is_unlocked,
            dao_wallets_count: dao_count,
            client_wallets_count: client_count,
        })
    }

    /// Executa backup atômico do banco de dados SQLite para o destino especificado
    pub fn export_backup(&self, destination_path: &std::path::Path) -> Result<u64, StorageError> {
        let conn = self.get_connection()?;
        // Executa checkpoint WAL para garantir consistência dos dados em disco
        conn.execute_batch("PRAGMA wal_checkpoint(TRUNCATE);")?;

        if let Some(parent) = destination_path.parent() {
            fs::create_dir_all(parent)?;
        }

        let bytes_copied = fs::copy(&self.db_path, destination_path)?;
        Ok(bytes_copied)
    }

    /// Restaura o banco de dados a partir de um arquivo de backup verificado
    pub fn restore_backup(&self, source_path: &std::path::Path) -> Result<(), StorageError> {
        if !source_path.exists() {
            return Err(StorageError::InvalidData(format!(
                "Arquivo de backup não encontrado: {:?}",
                source_path
            )));
        }

        // Valida que o arquivo de backup é um banco SQLite íntegro com a tabela vault_header
        let test_conn =
            Connection::open_with_flags(source_path, rusqlite::OpenFlags::SQLITE_OPEN_READ_ONLY)?;
        let has_vault_header: bool = test_conn
            .query_row(
                "SELECT count(*) FROM sqlite_master WHERE type='table' AND name='vault_header'",
                [],
                |row| row.get::<_, i64>(0),
            )
            .map(|count| count > 0)
            .unwrap_or(false);

        if !has_vault_header {
            return Err(StorageError::InvalidData(
                "Arquivo de backup inválido: tabela vault_header ausente.".to_string(),
            ));
        }
        drop(test_conn);

        // Remove banco ativo e arquivos WAL/SHM associados antes da substituição atômica
        let _ = fs::remove_file(&self.db_path);
        let wal_path = self.db_path.with_extension("sqlite-wal");
        let shm_path = self.db_path.with_extension("sqlite-shm");
        let _ = fs::remove_file(wal_path);
        let _ = fs::remove_file(shm_path);

        if let Some(parent) = self.db_path.parent() {
            fs::create_dir_all(parent)?;
        }

        fs::copy(source_path, &self.db_path)?;
        Ok(())
    }

    fn test_concurrent_multithread_initialization_race() {
        use std::sync::{Arc, Barrier};
        use std::thread;

        let test_dir =
            std::env::temp_dir().join(format!("devos_concurrency_race_{}", rand::random::<u32>()));
        let db = LocalDatabase::new(test_dir.clone());
        let conn = db.get_connection().unwrap();
        conn.execute_batch(
            "CREATE TABLE IF NOT EXISTS vault_header (id INTEGER PRIMARY KEY, version INTEGER NOT NULL, canary_envelope TEXT NOT NULL, created_at TEXT NOT NULL);"
        ).unwrap();
        drop(conn);

        const THREAD_COUNT: usize = 10;
        let barrier = Arc::new(Barrier::new(THREAD_COUNT));
        let mut handles = Vec::new();

        for i in 0..THREAD_COUNT {
            let b = Arc::clone(&barrier);
            let dir = test_dir.clone();
            let thread_password = format!("PasswordForThread{}!Secure123", i);

            let handle = thread::spawn(move || {
                let thread_db = LocalDatabase::new(dir);
                let header = VaultHeader::new(&thread_password).expect("Header gerado");
                // Todas as 10 threads aguardam aqui e disparam juntas
                b.wait();
                let res = thread_db.initialize_vault_if_absent(&header);
                (i, thread_password, res.is_ok())
            });

            handles.push(handle);
        }

        let mut success_count = 0;
        let mut winning_password = String::new();

        for h in handles {
            let (_i, pwd, is_ok) = h.join().unwrap();
            if is_ok {
                success_count += 1;
                winning_password = pwd;
            }
        }

        // Exatamente UMA thread pode ter sucesso
        assert_eq!(
            success_count, 1,
            "Exatamente UMA thread concorrente deve ter sucesso na inicialização atômica"
        );

        // O cabeçalho salvo no disco deve corresponder exatamente à thread vencedora
        let final_header = db.get_header().unwrap().expect("Header persistido");
        assert!(
            final_header.verify_password(&winning_password),
            "Header deve ser da thread vencedora"
        );

        let _ = fs::remove_dir_all(test_dir);
    }
```
